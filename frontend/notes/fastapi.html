FastAPI — бэкенд на Python

FastAPI — фреймворк для создания API на Python. Быстрый, современный, автоматически генерирует документацию.

Аналогия: API — это официант в ресторане. Ты (фронтенд) говоришь официанту что хочешь, он идёт на кухню (база данных) и приносит результат. Ты не знаешь как устроена кухня — тебе важен только результат.

Структура приложения

PYTHON backend/app/main.py 
from fastapi import FastAPI, HTTPException
# import — подключить библиотеку
# FastAPI — основной класс приложения
# HTTPException — возвращать ошибки с кодом (400, 500...)

from fastapi.middleware.cors import CORSMiddleware
# CORS — разрешить фронтенду обращаться к API с другого адреса

import psycopg2
# драйвер для подключения к PostgreSQL

import os
# встроенная библиотека Python для работы с переменными окружения

app = FastAPI(title="DevOps Portfolio API")
# создаём приложение

app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"],   # * = разрешить всем сайтам
    allow_methods=["*"],
    allow_headers=["*"],
)

Эндпоинты

Эндпоинт — адрес на который можно отправить запрос. Каждый эндпоинт делает одно конкретное действие.

PYTHON GET запросы — получить данные 
@app.get("/")
# @app.get — декоратор: при GET запросе на "/" вызвать функцию ниже
def root():
    return {"status": "ok"}
# return — вернуть JSON ответ

@app.get("/topics")
def get_topics():
    conn = get_db()          # открыть соединение с базой
    cur = conn.cursor()      # инструмент для SQL запросов
    cur.execute("SELECT id, name, progress, notes FROM topics")
    rows = cur.fetchall()    # получить все строки
    cur.close()
    conn.close()             # закрыть соединение
    return [
        {"id": r[0], "name": r[1], "progress": r[2], "notes": r[3]}
        for r in rows        # list comprehension — создать список из строк
    ]
PYTHON PUT запрос — обновить данные 
@app.put("/topics/{topic_id}")
# {topic_id} — переменная в URL, например /topics/3
def update_topic(topic_id: int, progress: int, notes: str = ""):
# topic_id: int — FastAPI проверит что это число
# notes: str = "" — необязательный параметр, по умолчанию пустая строка

    if progress < 0 or progress > 100:
        raise HTTPException(status_code=400, detail="Progress must be 0-100")
    # 400 — код ошибки "неправильный запрос"

    conn = get_db()
    cur = conn.cursor()
    cur.execute(
        "UPDATE topics SET progress=%s, notes=%s WHERE id=%s",
        (progress, notes, topic_id)
    )
    # %s — заполнитель для значений (защита от SQL инъекций)
    conn.commit()   # подтвердить изменения — без этого не сохранится
    cur.close()
    conn.close()
    return {"status": "updated"}
